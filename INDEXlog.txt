function saveNewCita(){
const cliente = $("#ci_cli").value.trim();
const fechaCita = $("#ci_fec").value;
const horaCita = $("#ci_hor").value;
const operador = $("#ci_op").value;
const eco = $("#ci_uni").value;

if(!cliente || !fechaCita) return alert("Falta el cliente o fecha");

}

function updateCitaTime(id, field, val){
const c = S.citas.find(x => x.id === id);
if(!c) return;
c[field] = val;
if(field === 'termino' && val) {
  CTAB = 'finalizadas';
}
save(); R();
}

function delCita(id){
if(!confirm("¿Eliminar esta cita?")) return;
S.citas = S.citas.filter(x => x.id !== id);
save(); R();
}

}
